<script>
    const bypassIdPerTest = true; 
    let idUtenteEstratto = "";      

    const urlParams = new URLSearchParams(window.location.search);
    const idIdentificato = urlParams.get('id') ? urlParams.get('id').trim() : '';

    if (idIdentificato) {
        idUtenteEstratto = idIdentificato;
        const urlPulito = window.location.protocol + "//" + window.location.host + window.location.pathname + window.location.hash;
        window.history.replaceState({ path: urlPulito }, '', urlPulito);
    } else if (bypassIdPerTest === true) {
        idUtenteEstratto = "TEST_LOCALE";
    }

    function switchTab(tabId) {
        const targetSection = document.getElementById(tabId);
        if (!targetSection) return;

        window.location.hash = tabId;

        const sections = document.querySelectorAll('.content-section');
        sections.forEach(section => section.classList.remove('active'));

        const buttons = document.querySelectorAll('.tab-btn');
        buttons.forEach(btn => btn.classList.remove('active'));

        targetSection.classList.add('active');

        const activeBtn = document.getElementById('btn-' + tabId);
        if (activeBtn) activeBtn.classList.add('active');

        if (tabId === 'noti') {
            caricaProblemiNotiPubblici();
        }
    }

    function gestisciAncoraURL() {
        if (window.location.hash) {
            const targetTab = window.location.hash.substring(1);
            if (document.getElementById(targetTab)) {
                switchTab(targetTab);
                return true;
            }
        }
        return false;
    }

    async function caricaProblemiNotiPubblici() {
        const container = document.getElementById("listaBugNoti");
        if (!container) return;

        try {
            const response = await fetch("segnalazioni.json?t=" + new Date().getTime(), {
                cache: "no-store",
                headers: {
                    "Cache-Control": "no-cache, no-store, must-revalidate",
                    "Pragma": "no-cache"
                }
            });
            if (!response.ok) throw new Error("File non trovato");

            const segnalazioni = await response.json();
            const listaSms = Object.values(segnalazioni);
            listaSms.sort((a, b) => new Date(b.dataInvio) - new Date(a.dataInvio));

            if (listaSms.length === 0) {
                container.innerHTML = "<p style='color:#2e7d32;'><b>Nessun bug critico rilevato! Il gioco è stabile.</b></p>";
                return;
            }

            let htmlIniezione = "";
            listaSms.forEach((item) => {
                let coloreStato = item.stato === "Risolto" ? "#2e7d32" : "#ff9800";
                let categoriaUmana = "Segnalazione Generale";
                if (item.categoria === "BugReport") categoriaUmana = "Bug Riscontrato";
                if (item.categoria === "AccountIssue") categoriaUmana = "Problema di Accesso / Salvataggio";
                if (item.categoria === "Feedback") categoriaUmana = "Suggerimento Tester";

                htmlIniezione += `
                    <div style="padding:15px; margin-bottom:12px; background:#f5f5f5; border-left:5px solid ${coloreStato}; border-radius:6px; color:#333;">
                        <h4 style="margin:0 0 5px 0; color:var(--primary-color); font-size:1.1em;">${categoriaUmana}</h4>
                        <p style="margin:5px 0 10px 0; font-size:0.95em; white-space: pre-wrap;">${item.messaggio}</p>
                        <div style="display:flex; justify-content:space-between; align-items:center; font-size:0.8em; color:#666; border-top:1px solid #e0e0e0; padding-top:8px;">
                            <span>Data: ${new Date(item.dataInvio).toLocaleString('it-IT')}</span>
                            <span style="font-weight:bold; color:${coloreStato}; text-transform:uppercase;">Stato: ${item.stato || 'In attesa'}</span>
                        </div>
                    </div>`;
            });
            container.innerHTML = htmlIniezione;
        } catch (error) {
            container.innerHTML = "<p style='color:#2e7d32;'><b>Nessun bug critico rilevato! Il gioco è stabile.</b></p>";
        }
    }

    document.addEventListener("DOMContentLoaded", function() {
        const form = document.getElementById("supportForm");
        if (!form) return;

        const submitBtn = document.getElementById("submitBtn");
        const statusBox = document.getElementById("statusBox");
        let isSubmitting = false;

        form.addEventListener("submit", function(e) {
            e.preventDefault();
            if (isSubmitting) return;

            if (!idUtenteEstratto) {
                statusBox.innerText = "Errore: Impossibile identificare l'utente dal gioco.";
                statusBox.style.color = "#d32f2f";
                statusBox.style.backgroundColor = "#ffebee";
                statusBox.style.border = "1px solid #d32f2f";
                statusBox.style.display = "block";
                return;
            }

            isSubmitting = true;
            submitBtn.disabled = true;
            submitBtn.innerText = "Salvataggio in corso...";

            const datiForm = {
                userId: idUtenteEstratto, 
                categoria: document.getElementById("categoria").value,
                messaggio: document.getElementById("messaggio").value.trim()
            };

            fetch("salva_segnalazione.php", {
                method: "POST",
                headers: { "Content-Type": "application/json" },
                body: JSON.stringify(datiForm)
            })
            .then(r => r.json())
            .then(data => {
                statusBox.style.display = "block";
                if(data.success) {
                    statusBox.innerText = "Segnalazione salvata con successo sul server!";
                    statusBox.style.color = "#2e7d32";
                    statusBox.style.backgroundColor = "#e8f5e9";
                    statusBox.style.border = "1px solid #2e7d32";
                    form.reset();
                } else {
                    statusBox.innerText = "Errore durante il salvataggio: " + (data.error || "Riprova.");
                    statusBox.style.color = "#d32f2f";
                    statusBox.style.backgroundColor = "#ffebee";
                    statusBox.style.border = "1px solid #d32f2f";
                }
            })
            .catch(err => {
                statusBox.innerText = "Errore di connessione al server.";
                statusBox.style.color = "#d32f2f";
                statusBox.style.backgroundColor = "#ffebee";
                statusBox.style.border = "1px solid #d32f2f";
                statusBox.style.display = "block";
            })
            .finally(() => {
                isSubmitting = false;
                submitBtn.disabled = false;
                submitBtn.innerText = "Invia Segnalazione";
            });
        });

        gestisciAncoraURL();
    });

    window.addEventListener('hashchange', gestisciAncoraURL);
</script>
